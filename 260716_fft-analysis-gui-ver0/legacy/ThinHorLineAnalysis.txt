    methods (Access = public)
        
        function [maxPeak2Valley, avgPeak2Valley, Dprofile, ActiveImage_avg, x_mm, start_x, end_x, start_y, end_y, x_cpcm, profile_fft, maxPeak, Dprofile_cut, peaks2valleys] ...
                = showData(app, ActiveImage, resize_V, resize_H, avgFiltSize, avgRefFiltSize, ROI_x, ROI_y,...
                           panel_long_mm, scanWidth_mm, cut_mmLow, cut_mmHigh, crop_left, crop_right, crop_top, crop_bottom)

            try

            ActiveImage_resized_temp = imresize(ActiveImage,[resize_V,resize_H]);
            ActiveImage_resized = ActiveImage_resized_temp(1+crop_top:end-crop_bottom, 1+crop_left:end-crop_right);
            hA = fspecial('average',[avgFiltSize, avgFiltSize]);
            ActiveImage_avg = imfilter(ActiveImage_resized,hA,'symmetric');
            hA_ref = fspecial('average',[avgRefFiltSize, avgRefFiltSize]);
            ActiveImage_ref = imfilter(ActiveImage_resized,hA_ref,'symmetric');

            %center_x = round(resize_H/2);
            %center_y = round(resize_V/2);
            center_x = round(size(ActiveImage_resized,2)/2);
            center_y = round(size(ActiveImage_resized,1)/2);
            start_x = floor(center_x-ROI_x/2+1);
            end_x = floor(center_x+ROI_x/2);
            start_y = floor(center_y-ROI_y/2+1);
            end_y = floor(center_y+ROI_y/2);
            
            profile = mean(ActiveImage_avg(start_y:end_y,start_x:end_x));
            profile_ref = mean(ActiveImage_ref(start_y:end_y,start_x:end_x));
            Dprofile = 100*(profile - profile_ref)./profile_ref;

            mmPP = panel_long_mm / resize_H;
            x_mm = mmPP*(1:size(profile,2));
            [peaks,locPeaks]=findpeaks(Dprofile, x_mm, 'MinPeakWidth',0.1); % 0.1mm 이상 Peak만 Search
            [valleys,locValleys]=findpeaks(-Dprofile, x_mm, 'MinPeakWidth',0.1); % 0.1mm 이상 Peak만 Search
            valleys=-valleys;

            peaks2valleys=zeros(size(locPeaks,2),1);
            maxPeak = 0;
            for k = 1:size(locPeaks,2)
                peak2valleys_temp = max(peaks(k)-valleys(find(abs(locPeaks(k)-locValleys)<scanWidth_mm,10,'first')));
                if isempty(peak2valleys_temp) == 0
                    peaks2valleys(k) = peak2valleys_temp;
                end
                if peaks2valleys(k) == max(peaks2valleys)
                    maxPeak = peaks(k);
                end
                % 각 Peak 기준 Scan Width 이내 가까운 2개 Valley를 찾고 2개 중 Peak to Valley 값이 큰 값을 선택
            end

            maxPeak2Valley = max(peaks2valleys);
            avgPeak2Valley = mean(peaks2valleys);

            peaks2valleys = sort(peaks2valleys,'descend');

            x_cpcm = (1:floor(ROI_x/2)-1)/(mmPP*ROI_x)*10;
            
            %임시 추가
            %Dprofile = imresize(Dprofile,[1,size(Dprofile,2)*3]);

            profile_fft_temp0 = fft(Dprofile);
            profile_fft_temp = abs(profile_fft_temp0);
            profile_fft = profile_fft_temp(2:floor(ROI_x/2));

            %profile_fft_temp(1);

             % 주파수 Cut 추가
            interval_cpcm = x_cpcm(1);
            
            profile_fft_temp0(2:round(10/(cut_mmHigh*interval_cpcm))) = 0;
            profile_fft_temp0(end-round(10/(cut_mmHigh*interval_cpcm))+2:end) = 0;
            profile_fft_temp0(round(10/(cut_mmLow*interval_cpcm))+2:end-round(10/(cut_mmLow*interval_cpcm))) = 0;

            Dprofile_cut = ifft(profile_fft_temp0);

            if app.show_plot == 1
                h=figure(4); set(h, "Name","Profile"), plot(x_mm,profile)
                hold on
                plot(x_mm,profile_ref)
                hold off

                profile_fft_temp1 = abs(profile_fft_temp0);
                profile_fft_temp2 = profile_fft_temp1(2:floor(ROI_x/2));
                %h=figure(5); set(h, "Name","FFT"), plot(x_cpcm(1:100), profile_fft_temp(2:101))
                %h=figure(6); set(h, "Name","FFT(cut)"), plot(x_cpcm(1:100), profile_fft_temp2(1:100))
                
                h=figure(5); set(h, "Name","FFT"), plot(x_cpcm(1:end), profile_fft(1:end))
                h=figure(6); set(h, "Name","FFT(cut)"), plot(x_cpcm(1:end), profile_fft_temp2(1:end))

                img_cut = imresize(Dprofile_cut,[ROI_y,ROI_x]);
                h=figure(7); set(h, "Name","ROI Wave(cut)"), imshow(img_cut,[])
            end

            catch ME
                message = {};
                for i = 1:size(ME.stack)
                    add_message = [ME.stack(i).name '(' num2str(ME.stack(i).line) '번 Line)'];
                    message = [message; add_message];
                end
                msgbox([ME.message; ' '; message],'Error');
            end
            
        end
    end

        % Button pushed function: Button_run
        function Button_runPushed(app, event)
            
            try

            
            % Excel Writing

            file_save = ['results_' erase(datestr(now),{'-', ' ', ':'}) '.xlsx'];
            xlswrite([app.directory '\' file_save], {'Please Wait...'}, 3, 'A1');
            Excel = actxserver('Excel.Application');
            Workbooks = Excel.Workbooks;
            Excel.Visible=1;
            Workbook=Workbooks.Open([app.directory '\' file_save]);
            Sheets = Excel.ActiveWorkBook.Sheets;
            sheet1 = get(Sheets, 'Item', 1);
            sheet2 = get(Sheets, 'Item', 2);
            sheet3 = get(Sheets, 'Item', 3);
            %invoke(sheet1, 'Activate');
            %invoke(sheet2, 'Activate');
            sheet1.Activate;
            sheet2.Activate;
            sheet3.Activate;
            %Activesheet = Excel.Activesheet;
